-- ============================================================ -- ANOTE — SQL 2 CORRIGIDO: FUNÇÕES CLIENTE -- (com DROP FUNCTION para evitar conflito de tipo de retorno) -- ============================================================ -- ========== cliente_cardapio ========== DROP FUNCTION IF EXISTS cliente_cardapio(uuid); CREATE OR REPLACE FUNCTION cliente_cardapio(p_store_id uuid) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_store record; BEGIN SELECT id, name, address, phone, paused INTO v_store FROM stores WHERE id = p_store_id LIMIT 1; IF v_store IS NULL THEN RETURN json_build_object('ok', false, 'error', 'Loja não encontrada'); END IF; RETURN json_build_object( 'ok', true, 'store', json_build_object( 'id', v_store.id, 'name', v_store.name, 'address', v_store.address, 'phone', v_store.phone, 'paused', v_store.paused ), 'categories', COALESCE(( SELECT json_agg(row_to_json(c)) FROM ( SELECT id, store_id, name, sort_order FROM categories WHERE store_id = p_store_id ORDER BY sort_order, name ) c ), '[]'::json), 'products', COALESCE(( SELECT json_agg(row_to_json(p)) FROM ( SELECT id, store_id, category_id, name, description, price, image_url, sort_order FROM products WHERE store_id = p_store_id AND is_active = true ORDER BY sort_order, name ) p ), '[]'::json) ); END; $$; -- ========== cliente_cadastrar ========== DROP FUNCTION IF EXISTS cliente_cadastrar(uuid, text, text); CREATE OR REPLACE FUNCTION cliente_cadastrar(p_store_id uuid, p_name text, p_phone text) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_id uuid; BEGIN SELECT id INTO v_id FROM customers WHERE store_id = p_store_id AND phone = p_phone LIMIT 1; IF v_id IS NULL THEN INSERT INTO customers (store_id, name, phone) VALUES (p_store_id, p_name, p_phone) RETURNING id INTO v_id; ELSE UPDATE customers SET name = p_name WHERE id = v_id; END IF; RETURN json_build_object('ok', true, 'customer', json_build_object('id', v_id, 'name', p_name, 'phone', p_phone)); END; $$; -- ========== cliente_pedido_status ========== DROP FUNCTION IF EXISTS cliente_pedido_status(uuid, uuid); CREATE OR REPLACE FUNCTION cliente_pedido_status(p_order_id uuid, p_customer_id uuid) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_order record; BEGIN SELECT * INTO v_order FROM orders WHERE id = p_order_id AND customer_id = p_customer_id LIMIT 1; IF v_order IS NULL THEN RETURN json_build_object('ok', false, 'error', 'Pedido não encontrado'); END IF; RETURN json_build_object( 'ok', true, 'status', v_order.status, 'order_number', v_order.order_number, 'total', v_order.total, 'items', COALESCE(( SELECT json_agg(json_build_object( 'qty', oi.qty, 'name', (SELECT name FROM products WHERE id = oi.product_id) )) FROM order_items oi WHERE oi.order_id = p_order_id ), '[]'::json) ); END; $$; -- ========== cliente_historico ========== DROP FUNCTION IF EXISTS cliente_historico(uuid, uuid); CREATE OR REPLACE FUNCTION cliente_historico(p_customer_id uuid, p_store_id uuid) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN RETURN json_build_object( 'ok', true, 'orders', COALESCE(( SELECT json_agg(row_to_json(o)) FROM ( SELECT ord.order_number, ord.status, ord.total, ord.created_at, ( SELECT COALESCE(json_agg(json_build_object( 'qty', oi.qty, 'name', (SELECT name FROM products WHERE id = oi.product_id) )), '[]'::json) FROM order_items oi WHERE oi.order_id = ord.id ) AS items FROM orders ord WHERE ord.customer_id = p_customer_id AND ord.store_id = p_store_id ORDER BY ord.created_at DESC ) o ), '[]'::json) ); END; $$; -- ========== cliente_chamar ========== DROP FUNCTION IF EXISTS cliente_chamar(uuid, uuid, text); CREATE OR REPLACE FUNCTION cliente_chamar(p_store_id uuid, p_table_id uuid, p_type text) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ DECLARE v_id uuid; BEGIN INSERT INTO alerts (store_id, table_id, type) VALUES (p_store_id, p_table_id, p_type) RETURNING id INTO v_id; RETURN json_build_object('ok', true, 'id', v_id); END; $$; -- ========== cliente_meus_alertas ========== DROP FUNCTION IF EXISTS cliente_meus_alertas(uuid, uuid); CREATE OR REPLACE FUNCTION cliente_meus_alertas(p_store_id uuid, p_table_id uuid) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN RETURN json_build_object( 'ok', true, 'alerts', COALESCE(( SELECT json_agg(row_to_json(a)) FROM ( SELECT id, store_id, table_id, type, status, created_at FROM alerts WHERE store_id = p_store_id AND table_id = p_table_id ORDER BY created_at DESC ) a ), '[]'::json) ); END; $$; -- ========== cliente_fechar_alerta ========== DROP FUNCTION IF EXISTS cliente_fechar_alerta(uuid, uuid); CREATE OR REPLACE FUNCTION cliente_fechar_alerta(p_store_id uuid, p_alert_id uuid) RETURNS json LANGUAGE plpgsql SECURITY DEFINER AS $$ BEGIN DELETE FROM alerts WHERE id = p_alert_id AND store_id = p_store_id; RETURN json_build_object('ok', true); END; $$;
